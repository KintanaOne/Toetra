"""
Invalid CST mutation entrypoint.

This module composes all CST corruption strategies:

- grammar corruption
- parse corruption
- syntax corruption
- token corruption
- tree corruption

GOAL:
    Stress parser robustness and AST builder resilience.

PIPELINE LEVEL:
    TEXT
      ↓
    TOKENS
      ↓
    CST / ParseTree
      ↓
    AST BUILDING

These mutations operate ONLY on Lark CST structures.
"""

from __future__ import annotations

import random

from hypothesis import strategies as st

from lark import Tree

from test.hypothesis.mutations.cst.grammar import (
    reorder_grammar_rules,
    remove_random_subtree,
)

from test.hypothesis.mutations.cst.parse import (
    corrupt_tree_data,
)

from test.hypothesis.mutations.cst.syntax import (
    remove_random_token,
)

from test.hypothesis.mutations.cst.tokens import (
    corrupt_token_type,
    corrupt_token_value,
)

from test.hypothesis.mutations.cst.tree import (
    duplicate_random_subtree,
    reverse_tree_children,
)

# =========================================================
# Registry
# =========================================================

INVALID_CST_MUTATIONS = [

    # grammar corruption
    reorder_grammar_rules,
    remove_random_subtree,

    # parse corruption
    corrupt_tree_data,

    # syntax corruption
    remove_random_token,

    # token corruption
    corrupt_token_type,
    corrupt_token_value,

    # tree corruption
    duplicate_random_subtree,
    reverse_tree_children,
]

# =========================================================
# Core mutation engine
# =========================================================


def apply_invalid_cst(
    cst: Tree,
    mutation_count: int = 1,
) -> Tree:
    """
    Apply N random CST mutations.
    """

    mutated = cst

    for _ in range(mutation_count):

        mutation_fn = random.choice(
            INVALID_CST_MUTATIONS
        )

        mutated = mutation_fn(mutated)

    return mutated


# =========================================================
# Hypothesis strategy
# =========================================================


@st.composite
def invalid_cst_program(
    draw,
    base_program,
) -> Tree:
    """
    Generate an invalid FORML CST.

    INPUT:
        Valid ParseTree strategy.

    OUTPUT:
        Corrupted CST / ParseTree.
    """

    program = draw(base_program)

    mutation_count = draw(
        st.integers(
            min_value=1,
            max_value=3,
        )
    )

    # heavy corruption mode
    if draw(st.booleans()):

        mutation_count *= 2

    return apply_invalid_cst(
        program,
        mutation_count,
    )



===== test\hypothesis\strategies\invalid\invalid_program.py =====

"""
Multi-layer invalid FORML program generation.

Pipeline:
    valid_program (string)
        ↓
    lexical invalid strategy
        ↓
    CST
        ↓
    structural invalid strategy
        ↓
    AST
        ↓
    semantic invalid strategy

This module orchestrates invalid generation across
FORML pipeline layers.
"""

import random
from hypothesis.strategies import composite

from dsl.parser.parser import parse_forml_code
from dsl.builder.program import parse_program

from test.hypothesis.strategies.valid.program_string import valid_string_program

# ================================
# STRATEGY LAYERS (NOT MUTATIONS)
# ================================

from test.hypothesis.strategies.invalid.invalid_string import invalid_string_program
from test.hypothesis.strategies.invalid.invalid_cst import invalid_cst_program
from test.hypothesis.strategies.invalid.invalid_ast import invalid_ast_program


# =========================================================
# MAIN STRATEGY
# =========================================================

@composite
def invalid_program(
    draw,
    string_ratio: float = 0.4,
    cst_ratio: float = 0.3,
    ast_ratio: float = 0.3,
):
    """
    Generate invalid FORML programs across multiple layers.

    Each layer has its own invalid strategy responsible
    for producing controlled corruption.
    """

    assert 0 <= string_ratio <= 1, "string_ratio must be in [0, 1]"
    assert 0 <= cst_ratio <= 1, "cst_ratio must be in [0, 1]"
    assert 0 <= ast_ratio <= 1, "ast_ratio must be in [0, 1]"
    assert (string_ratio + cst_ratio + ast_ratio) <= 1, "Total ratio must be <= 1"

    # -----------------------------------------------------
    # STEP 1: valid program
    # -----------------------------------------------------

    program = draw(valid_string_program())

    # -----------------------------------------------------
    # STEP 2: string corruption
    # -----------------------------------------------------

    if random.random() < string_ratio:
        program = draw(invalid_string_program(program))

    # -----------------------------------------------------
    # STEP 3: parse CST
    # -----------------------------------------------------

    try:
        cst = parse_forml_code(program)
    except Exception:
        return program

    # -----------------------------------------------------
    # STEP 4: cst corruption
    # -----------------------------------------------------

    if random.random() < cst_ratio:
        cst = draw(invalid_cst_program(cst))

    # -----------------------------------------------------
    # STEP 5: parse AST
    # -----------------------------------------------------

    try:
        ast = parse_program(cst)
    except Exception:
        return cst

    # -----------------------------------------------------
    # STEP 6: ast corruption
    # -----------------------------------------------------

    if random.random() < ast_ratio:
        ast = draw(invalid_ast_program(ast))

    return ast

def apply_invalid_lexical(text: str, mutation_count: int = 1) -> str:
    """
    Apply N random lexical mutations.
    """
    mutated = text

    for _ in range(mutation_count):
        mutation = random.choice(INVALID_STRING_MUTATIONS)
        mutated = mutation(mutated)

    return mutated

@st.composite
def invalid_string_program(draw, base_program : str) -> str:
    """
    Generate a lexically invalid FORML program.

    Parameters
    ----------
    base_program:
        strategy producing valid FORML programs (string)

    Returns
    -------
    str (invalid FORML program)
    """

    program = draw(base_program)

    # number of mutations (controlled chaos)
    mutation_count = draw(st.integers(min_value=1, max_value=3))

    # optional: probability of "heavy corruption"
    if draw(st.booleans()):
        mutation_count *= 2

    return invalid_string_program(program, mutation_count)
